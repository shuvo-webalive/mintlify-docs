```python Python
            client_id=os.environ["EXSITED_CLIENT_ID"],
            client_secret=os.environ["EXSITED_CLIENT_SECRET"],
            redirect_uri=os.environ["EXSITED_REDIRECT_URI"],
```

```java Java
import com.exsited.sdk.client.ClientRegistry;
import com.exsited.sdk.config.AuthConfig;
import com.exsited.sdk.config.SdkConfig;
import java.util.List;

public class Main {
    public static void main(String[] args) throws Exception {
        ClientRegistry.registerInstances(List.of(
                SdkConfig.builder(AuthConfig.builder()
                        .clientName("default")
                        .baseUrl(System.getenv("EXSITED_BASE_URL"))
                        .clientId(System.getenv("EXSITED_CLIENT_ID"))
                        .clientSecret(System.getenv("EXSITED_CLIENT_SECRET"))
                        .redirectUri(System.getenv("EXSITED_REDIRECT_URI"))
                        .build())
                        .raiseOnError(true)
                        .build()));
    }
}
```

```go Go
package main

import (
	"log"
	"os"

	exsited "bitbucket.org/webalive/exsited-sdk-go"
)

func main() {
	err := exsited.RegisterInstances(exsited.SdkConfig{
		Auth: exsited.AuthConfig{
			ClientName:   "default",
			BaseURL:      os.Getenv("EXSITED_BASE_URL"),
			ClientID:     os.Getenv("EXSITED_CLIENT_ID"),
			ClientSecret: os.Getenv("EXSITED_CLIENT_SECRET"),
			RedirectURI:  os.Getenv("EXSITED_REDIRECT_URI"),
		},
		RaiseOnError: true,
	})
	if err != nil {
		log.Fatal(err)
	}
}
```
